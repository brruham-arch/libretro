cmake_minimum_required(VERSION 3.18)
project(nes C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_C_FLAGS_DEBUG "-O2 -DNDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG")
file(GLOB LUA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/lua/src/*.c)
list(FILTER LUA_SRC EXCLUDE REGEX "/(lua|luac|onelua)\\.c$")
if(NOT LUA_SRC)
  message(FATAL_ERROR "Lua belum diunduh (build lewat GitHub Actions)")
endif()
add_library(nes SHARED nes.cpp ${LUA_SRC})
target_include_directories(nes PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lua/src)
if(CMAKE_ANDROID_ARCH_ABI STREQUAL "armeabi-v7a")
  target_compile_options(nes PRIVATE -marm)
endif()
target_link_libraries(nes log)
